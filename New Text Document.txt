function Invoke-GitWithTimeout {
    param (
        [string]$Arguments,
        [int]$TimeoutSeconds = 4
    )
    $retry = $true
    while ($retry) {
        Write-Host "Executing: git $Arguments" -ForegroundColor Cyan
        
        $pinfo = New-Object System.Diagnostics.ProcessStartInfo
        $pinfo.FileName = "git"
        $pinfo.Arguments = $Arguments
        $pinfo.UseShellExecute = $false
        $pinfo.RedirectStandardOutput = $true
        $pinfo.RedirectStandardError = $true
        $pinfo.CreateNoWindow = $true
        
        $process = New-Object System.Diagnostics.Process
        $process.StartInfo = $pinfo
        $process.Start() | Out-Null
        
        # Wait for the specified timeout
        $completed = $process.WaitForExit($TimeoutSeconds * 1000)
        
        if ($completed) {
            $output = $process.StandardOutput.ReadToEnd()
            $err = $process.StandardError.ReadToEnd()
            if ($output) { Write-Host $output }
            if ($err -match "fatal|error") { Write-Host $err -ForegroundColor Red }
            $retry = $false
        } else {
            Write-Host "Process hanging (> $TimeoutSeconds sec). Killing and retrying immediately..." -ForegroundColor Yellow
            $process.Kill()
            Start-Sleep -Milliseconds 500
        }
    }
}

foreach ($item in $items) {
    # Clean up any rogue quotes from Git status output
    $cleanItem = $item.Trim().Trim('"')
    
    # Append wildcard ONLY for folders
    $addPath = $cleanItem
    if ($addPath.EndsWith("/")) {
        $addPath = "$addPath*"
    }

    Write-Host "`n--- Processing: $cleanItem ---" -ForegroundColor Green
    
    Invoke-GitWithTimeout -Arguments "reset"
    Invoke-GitWithTimeout -Arguments "add `"$addPath`""
    Invoke-GitWithTimeout -Arguments "commit -m `"add $cleanItem`" --no-verify"
    Invoke-GitWithTimeout -Arguments "push --no-verify"
}
